### صفحة [ 1 ]

&lt;برنامج&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">برنامج</span> &lt;اسم_برنامج&gt; &lt;فاصلة منقوطة&gt; &lt;كتلة_برمجية&gt; &lt; نقطة&gt;

&lt;كتلة_برمجية&gt; := [&lt;جزء_التعريفات&gt;] &lt;قائمة_التعليمات&gt;

&lt;جزء_التعريفات&gt; := [&lt;تعريف_الثوابت&gt;] [&lt;تعريف_الانواع&gt;] [&lt;تعريف_المتغيرات&gt;] [&lt; تعريف_الاجراءات&gt;]

&lt;تعريف_الثوابت&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">ثابت</span> &lt; تعريف_ثابت&gt; (&lt; تعريف_ثابت&gt;)*

&lt;تعريف_ثابت&gt; := &lt;اسم_ثابت&gt; &lt;علامة_يساوي&gt; &lt; قيمة_ثابت&gt; &lt;فاصلة_منقوطة&gt;

&lt;تعريف_الانواع&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">نوع</span> &lt; تعريف_نوع&gt; ( &lt;تعريف_نوع&gt;)*

&lt;تعريف_نوع&gt; := &lt;اسم_نوع&gt; &lt;علامة_يساوي&gt; &lt;نوع_مركب&gt; &lt;فاصلة_منقوطة&gt;

&lt;نوع_مركب&gt; := &lt;نوع_قائمة&gt; | &lt;نوع_سجل&gt;

&lt;نوع_قائمة&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">قائمة</span> &lt;قوس_مربع_يمين&gt; &lt;عدد&gt; &lt; قوس_مربع_يسار&gt; <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">من</span> &lt;نوع_بيانات&gt;

<hr style="border: 1px dashed #ccc; margin: 25px 0;">

### صفحة [ 2 ]

&lt;نوع_سجل&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">سجل</span> &lt;قوس_مجموعة_يمين&gt; &lt;قائمة_حقول&gt; &lt;قوس_مجموعة_يسار&gt;

&lt;قائمة_حقول&gt; := &lt;تعريف_حقل&gt; (&lt;فاصلة_منقوطة&gt; &lt;تعريف_حقل&gt;)*

&lt;تعريف_حقل&gt; := &lt;اسم_حقل&gt; (&lt;فاصلة&gt; &lt;معرف&gt;)* &lt;نقطتين&gt; &lt;نوع_بيانات&gt;

&lt;تعريف_المتغيرات&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">متغير</span> &lt;تعريف_متغير&gt; (&lt;تعريف_متغير&gt;)*

&lt;تعريف_متغير&gt; := &lt;مجموعة_متغيرات&gt; &lt;فاصلة_منقوطة&gt;

&lt;مجموعة_متغيرات&gt; := &lt;اسم_متغير&gt; (&lt;فاصلة&gt; &lt;اسم_متغير&gt;)* &lt;نقطتين&gt; &lt;نوع_بيانات&gt;

&lt;تعريف_الاجراءات&gt; := &lt;تعريف_اجراء&gt; (&lt;تعريف_اجراء&gt;)*

&lt;تعريف_اجراء&gt; := &lt;رأس_اجراء&gt; &lt;كتلة_اجراء&gt; &lt;فاصلة_منقوطة&gt;

&lt;رأس_اجراء&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">اجراء</span> &lt;اسم_اجراء&gt; &lt;قوس_يمين&gt; [&lt;قائمة_معلمات_شكلية&gt;] &lt;قوس_يسار&gt; &lt;فاصلة_منقوطة&gt;

&lt;كتلة_اجراء&gt; := &lt;كتلة_برمجية&gt;

&lt;قائمة_معلمات_شكلية&gt; := &lt;تعريف_معلمة&gt; (&lt;فاصلة_منقوطة&gt; &lt;تعريف_معلمة&gt;)*

&lt;تعريف_معلمة&gt; := [<span style="color: #0056b3; font-weight: bold; text-decoration: underline;">بالقيمة</span> | <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">بالمرجع</span>] &lt;مجموعة_متغيرات&gt;

&lt;نوع_بيانات&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">صحيح</span> | <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">حقيقي</span> | <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">منطقي</span> | <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">حرفي</span> | <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">خيط_رمزي</span> | &lt;اسم_نوع&gt;

<hr style="border: 1px dashed #ccc; margin: 25px 0;">

### صفحة [ 3 ]

&lt;قائمة_تعليمات&gt; := &lt;قوس_مجموعة_يمين&gt; &lt;تعليمة&gt; (&lt;فاصلة_منقوطة&gt; &lt;تعليمة&gt;)* &lt;قوس_مجموعة_يسار&gt;

&lt;تعليمة&gt; := &lt;تعليمة_اسناد&gt; | &lt;تعليمة_ادخال&gt; | &lt;تعليمة_اخراج&gt; | &lt;تعليمة_استدعاء&gt; | &lt;تعليمة_شرط&gt; | &lt;تعليمة_تكرار&gt; | &lt;قائمة_تعليمات&gt; | &lt;فراغ&gt;

&lt;تعليمة_اسناد&gt; := &lt;متغير_وصول&gt; &lt;علامة_يساوي&gt; &lt;تعبير&gt;

&lt;تعليمة_ادخال&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">اقرا</span> &lt;قوس_يمين&gt; &lt;متغير_وصول&gt; &lt;قوس_يسار&gt;

&lt;تعليمة_اخراج&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">اطبع</span> &lt;قوس_يمين&gt; &lt;قائمة_طباعة&gt; &lt;قوس_يسار&gt;

&lt;قائمة_طباعة&gt; := &lt;عنصر_طباعة&gt; (&lt;فاصلة&gt; &lt;عنصر_طباعة&gt;)*

&lt;عنصر_طباعة&gt; := &lt;متغير_وصول&gt; | &lt;حرفي&gt;

&lt;تعليمة_استدعاء&gt; := &lt;اسم_اجراء&gt; &lt;قوس_يمين&gt; [&lt;قائمة_معلمات_حقيقية&gt;] &lt;قوس_يسار&gt;

&lt;قائمة_معلمات_حقيقية&gt; := &lt;معلمة_حقيقية&gt; (&lt;فاصلة&gt; &lt;معلمة_حقيقية&gt;)*

&lt;معلمة_حقيقية&gt; := &lt;تعبير&gt; | &lt;متغير_وصول&gt;

&lt;تعليمة_شرط&gt; := &lt;تعليمة_اذا&gt; | &lt;تعليمة_اذا_و_الا&gt; | &lt;تعليمة_اذا_و_الا_اذا&gt;

&lt;تعليمة_اذا&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">اذا</span> &lt;قوس_يمين&gt; &lt;شرط&gt; &lt;قوس_يسار&gt; <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">فان</span> &lt;تعليمة&gt;

<hr style="border: 1px dashed #ccc; margin: 25px 0;">

### صفحة [ 4 ]

&lt;شرط&gt; := &lt;تعبير&gt;

&lt;تعليمة_اذا_و_الا&gt; := &lt;تعليمة_اذا&gt; <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">والا</span> &lt;تعليمة&gt;

&lt;تعليمة_اذا_و_الا_اذا&gt; := &lt;تعليمة_اذا&gt; &lt;فاصلة_منقوطة&gt; <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">والا</span> &lt;تعليمة_اذا&gt; (&lt;فاصلة_منقوطة&gt; <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">والا</span> &lt;تعليمة_اذا&gt;)* [<span style="color: #0056b3; font-weight: bold; text-decoration: underline;">والا</span> &lt;تعليمة&gt;]

&lt;تعليمة_تكرار&gt; := &lt;تعليمة_كرر&gt; | &lt;تعليمة_كرر_طالما&gt; | &lt;تعليمة_كرر_حتى&gt;

&lt;تعليمة_كرر&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">كرر</span> &lt;قوس_يمين&gt; &lt;مجال_التكرار&gt; &lt;قوس_يسار&gt; &lt;تعليمة&gt;

&lt;مجال_التكرار&gt; := &lt;اسم_متغير&gt; &lt;علامة_يساوي&gt; &lt;تعبير&gt; <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">الى</span> &lt;تعبير&gt; [<span style="color: #0056b3; font-weight: bold; text-decoration: underline;">اضف</span> &lt;تعبير&gt;]

&lt;تعليمة_كرر_طالما&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">طالما</span> &lt;قوس_يمين&gt; &lt;شرط&gt; &lt;قوس_يسار&gt; <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">استمر</span> &lt;تعليمة&gt;

&lt;تعليمة_كرر_حتى&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">اعد</span> &lt;تعليمة&gt; <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">حتى</span> &lt;قوس_يمين&gt; &lt;شرط&gt; &lt;قوس_يسار&gt;

&lt;تعبير&gt; := &lt;تعبير_بسيط&gt; [&lt;معامل_ربط&gt; &lt;تعبير_بسيط&gt;]

&lt;تعبير_بسيط&gt; := [&lt;معامل_اشارة&gt;] &lt;حد&gt; (&lt;معامل_جمع&gt; &lt;حد&gt;)*

&lt;معامل_ربط&gt; := &lt; | &gt; | &lt;= | &gt;= | == | !=

&lt;معامل_اشارة&gt; := + | -

<hr style="border: 1px dashed #ccc; margin: 25px 0;">

### صفحة [ 5 ]

&lt;معامل_جمع&gt; := &lt;معامل_اشارة&gt; | ||

&lt;حد&gt; := &lt;عامل&gt; (&lt;معامل_ضرب&gt; &lt;عامل&gt;)*

&lt;معامل_ضرب&gt; := ^ | * | / | \ | % | &amp;&amp;

&lt;عامل&gt; := &lt;قيمة_ثابت&gt; | &lt;متغير_وصول&gt; | &lt;قوس_يمين&gt; &lt;تعبير&gt; &lt;قوس_يسار&gt; | ! &lt;عامل&gt;

&lt;متغير_وصول&gt; := &lt;اسم_متغير&gt; | &lt;اسم_نوع&gt; &lt;مختار&gt;

&lt;مختار&gt; := &lt;مختار_مفهرس&gt; | &lt;حقل_مختار&gt;

&lt;مختار_مفهرس&gt; := &lt;قوس_مربع_يمين&gt; &lt;تعبير&gt; &lt;قوس_مربع_يسار&gt;

&lt;حقل_مختار&gt; := &lt;نقطة&gt; &lt;اسم_حقل&gt;

&lt;قيمة_ثابت&gt; := &lt;رقمي&gt; | &lt;حرفي&gt; | &lt;منطقي&gt; | &lt;اسم_ثابت&gt;

&lt;رقمي&gt; := &lt;رقم_صحيح&gt; | &lt;رقم_حقيقي&gt;

&lt;رقم_صحيح&gt; := &lt;رقم&gt; (&lt;رقم&gt;)*

&lt;رقم_حقيقي&gt; := &lt;رقم_صحيح&gt; &lt;نقطة&gt; &lt;رقم_صحيح&gt;

&lt;حرفي&gt; := &lt;سلسلة_رمزية&gt; | &lt;رمز_مفرد&gt;

<hr style="border: 1px dashed #ccc; margin: 25px 0;">

### صفحة [ 6 ]

&lt;سلسلة_رمزية&gt; := &lt;علامة_تنصيص_مزدوجة&gt; (&lt;رمز&gt;)* &lt;علامة_تنصيص_مزدوجة&gt;

&lt;رمز_مفرد&gt; := &lt;علامة_تنصيص_يمين&gt; (&lt;رمز&gt;) &lt;علامة_تنصيص_يسار&gt;

&lt;منطقي&gt; := <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">صح</span> | <span style="color: #0056b3; font-weight: bold; text-decoration: underline;">خطأ</span>

&lt;اسم_برنامج&gt; := &lt;معرف&gt;

&lt;اسم_ثابت&gt; := &lt;معرف&gt;

&lt;اسم_نوع&gt; := &lt;معرف&gt;

&lt;اسم_حقل&gt; := &lt;معرف&gt;

&lt;اسم_متغير&gt; := &lt;معرف&gt;

&lt;اسم_اجراء&gt; := &lt;معرف&gt;

&lt;معرف&gt; := &lt;حرف&gt; (&lt;حرف&gt; | &lt;رقم&gt;)*

&lt;رقم&gt; := 0 | 1 | .. | 9

&lt;حرف&gt; := ا | ب | .. | ي

&lt;رمز&gt; := كل الرموز المستخدمة في النظام

    <li><b>رمز:</b> تعني كل الرموز المستخدمة في النظام (مباشرة من لوحة المفاتيح او عن طريق الاختصارات).</li>
